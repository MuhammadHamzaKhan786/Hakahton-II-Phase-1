#!/usr/bin/env python3
"""
Simple script to demonstrate the Todo application usage
"""
print("HOW TO USE THE TODO APPLICATION")
print("===============================")
print()
print("1. RUN THE APPLICATION:")
print("   python -m src.cli.todo_app")
print()
print("2. THE APPLICATION WILL SHOW A MENU LIKE THIS:")
print("   === Todo App ===")
print("   1. Add Task")
print("   2. View Tasks")
print("   3. Update Task")
print("   4. Delete Task")
print("   5. Toggle Completion")
print("   6. Exit")
print("   ================")
print()
print("3. AVAILABLE FEATURES:")
print()
print("   ADD TASK (option 1):")
print("   - Enter a task title when prompted")
print("   - The app will assign a unique ID automatically")
print("   - Example: 'Buy groceries' -> becomes task #1")
print()
print("   VIEW TASKS (option 2):")
print("   - Shows all tasks with their ID, title and status")
print("   - Example: 'ID: 1 | Title: Buy groceries | Status: Pending'")
print()
print("   UPDATE TASK (option 3):")
print("   - Enter the task ID you want to update")
print("   - Enter the new title for that task")
print("   - Example: Change 'Buy groceries' to 'Buy weekly groceries'")
print()
print("   DELETE TASK (option 4):")
print("   - Enter the task ID you want to delete")
print("   - The task will be permanently removed from memory")
print("   - Example: Delete task #1")
print()
print("   TOGGLE COMPLETION (option 5):")
print("   - Enter the task ID you want to mark/unmark")
print("   - If pending, it becomes completed")
print("   - If completed, it becomes pending again")
print("   - Example: Task #1 'Buy groceries' goes from [○] to [✓]")
print()
print("   EXIT (option 6):")
print("   - Safely closes the application")
print()
print("4. ERROR HANDLING:")
print("   - Invalid inputs are handled gracefully")
print("   - Non-existent task IDs show helpful error messages")
print("   - Empty titles are rejected with error messages")
print()
print("5. IMPORTANT NOTES:")
print("   - All data is stored in memory only")
print("   - Tasks are lost when the application closes")
print("   - Perfect for quick task management without persistence")
print()
print("TO START THE APP NOW, RUN: python -m src.cli.todo_app")
